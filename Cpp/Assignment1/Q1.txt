#include <iostream>
using namespace std;


//      get the status from the range
int status(double reading){
		int statuscode;
	if(reading < 0){statuscode = -1;}
	else if(reading < 30){statuscode = 0;}
	else if(reading < 45){statuscode = 1;}
	else if(reading < 60){statuscode = 2;}
	else{statuscode = 3;}
	//cout<<statuscode;

	return statuscode;
}

// get the status label from the  status code
void statuslabel(int statuscode){
	  switch (statuscode) {
		          case -1:
		              cout << "SENSOR_ERROR" << endl;
		              cout << "Action      : Sensor fault — check wiring" << endl;
		              break;

		          case 0:
		        	  cout << "NORMAL" << endl;
		        	  cout << "Action     : No action required" << endl;
		              break;

		          case 1:
		        	  cout << "WARNING " << endl;
		        	  cout << "Action      : Alert sent to supervisor" << endl;
		              break;

		          case 2:
		        	  cout << "CRITICAL " << endl;
		        	  cout << "Action      : Cooling system triggered" << endl;
		          	  break;

		          default:
		        	  cout << "SHUTDOWN " << endl;
		        	  cout << "Action      : Emergency shutdown initiated" << endl;
		      }


}


int main1() {
	double reading;

	cout << "Reading taken from the sensor" << endl;
	cin >> reading ;
	int statuscode = status(reading);  //give the code

	 float F = (reading*9.0/5.0)+32;

	cout << "Temperature : "<<reading<<" C  / "<< F <<" F"<<endl;

	cout << "Status      : ";  statuslabel(statuscode);     //will print the status label

    cout << "Reading     : "<< (reading<25 ?"Below average":"Above average") <<endl;


	return 0;
}
